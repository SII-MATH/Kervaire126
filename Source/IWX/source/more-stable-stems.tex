\begin{abstract}
\rev{Using techniques in motivic homotopy theory, especially the theorem of Gheorghe, the second and the third author on the isomorphism between motivic Adams spectral sequence for $C\tau$ and the algebraic Novikov spectral sequence for $BP_*$,
we compute the classical and motivic stable homotopy groups of spheres 
from dimension 0 to 90},
except for
some carefully enumerated uncertainties.
\end{abstract}
