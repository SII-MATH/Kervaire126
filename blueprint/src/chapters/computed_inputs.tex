\section{Computed Adams-page assertions}
\label{chap:computed-inputs}

This section states the coordinate, product, differential, and cofiber-map
assertions needed for the sphere and Hopf-cofiber argument, together with the
tmf comparisons. The assertions concern the actual Adams towers, using the
specified common coordinate systems of
Definition~\ref{def:computation-fixed-bindings}. Their mathematical
certification remains unfinished.

\subsection{Common page and coordinate notation}
\label{sec:computation-master-inventory}

Use $P,R,L,G,U,I,H,M$ and $C_X^{s,t}$ as in
Definition~\ref{def:computation-fixed-bindings}. Write
$A(s,t)=A^{s,t}$ and $\mathcal R_H$ for the specified mod--$2$ ring
structure on $H$. The symbol $\operatorname{atom}(a)$ denotes the
specified homogeneous sphere quotient generator named by $a$.
The sphere and tmf presentations, their monomial lists, and the staircase
and differential coordinates are those of Lin--Wang--Xu,
Zenodo record 14875701, version v126.3.cw49; see LWXMachine,
Sections~2--3. The selected sphere and cofiber claims include the equations
and page conditions used in the paper's Section~7 and Appendix.
Every statement below uses the same $P,R,U$ and actual source objects.

The selected spectra are $S^0$ and
$C\nu=\operatorname{cofib}(\nu:S^3\to S^0)$, with the specified Hopf
map. Their coordinate vectors are $R_{o,s,t,i}\in E_2^{s,t}(o)$.
The partial coordinate decoder $\operatorname{dec}_R(o,s,t,I)$ first
requires a matching entry of the fixed degree list, strictly increasing
indices, and every index below that entry's monomial-list length. On
success its value is $\sum_{i\in I}R_{o,s,t,i}$; otherwise it is undefined.
In particular, an empty coordinate list at a missing degree does not decode.

Write $\operatorname{CSV}_{s,t}(x,I)$ if there is a list of actual rows
of the fixed sphere CSV basis table whose local-index list is exactly $I$,
every row has degree $(s,t)$, and $|x|$ is the sum of those rows' literal
monomial values. This uses the table's prescribed row membership and
degree, rather than an independently chosen coordinate map.
For either actual classical sequence, write $\operatorname{rep}_r(x,u)$
if $r\ge2$ and one raw cycle at index $r-2$ has initial $E_2$ quotient
$x$ and $r$-page quotient $u$. Define the differential relation
\[
  \mathcal D_r^o(p,q;x,y)\quad\Longleftrightarrow\quad
  \begin{gathered}
    q=p+\operatorname{diffDeg}(r),\quad
    \exists u\in E_r^p(o),\ v\in E_r^q(o),\\
    \operatorname{rep}_r(x,u)\ \land\operatorname{rep}_r(y,v)
       \ \land\ d_ru=v,
  \end{gathered}
\]
where the last equality uses that actual degree transport.
Also write $\operatorname{Reach}_r^o(p,x)$ for
$\exists u\in E_r^p(o),\operatorname{rep}_r(x,u)$, and
$\operatorname{Bound}_r^o(p,x)$ for $r\ge2$ and existence of an element
of the actual raw boundary subobject at index $r-1$ whose image under
the fixed inclusions and initial $E_2$ quotient is $x$.
These definitions allow zero page classes and zero differential targets.
Reaching $E_7$ does not determine $d_7$. Reaching any finite page,
including $E_{1000}$ for the level-$9000$ condition, does not rule out
incoming boundaries and does not imply nonzero permanent survival.


\subsection{Sphere coordinates and bounded products}

\begin{theorem}[Sphere coordinate vectors have their prescribed monomial values]
  \label{thm:computation-sphereBasis-csv-values}
  \lean{KIP126.Challenge2.SphereBasisInterface.csv_values,KIP126.Challenge2.ComputationResults.sphereBasis}
  \notready
  \uses{def:computation-fixed-bindings}
  Let $\chi_{s,t}:E_2^{s,t}(S^0)\cong
  (\mathrm{BasisIndex}(s,t)\to_0\mathbb F_2)$ be the integer-linear
  coordinate equivalence with the specified coordinate vectors. For all $s,t\in\mathbb N$ with
  $t\le261$ and every $i\in\mathrm{BasisIndex}(s,t)$,
  \[
    \left|P_{s,t}^{-1}\bigl(\chi_{s,t}^{-1}(\delta_i)\bigr)\right|
        =\mathrm{basisValue}(\mathrm{basisRowAt}(s,t,i)),
  \]
  where $\delta_i$ is the finite-support coordinate vector with value
  one at $i$ and zero elsewhere. The equivalences and prescribed monomial
  values in this assertion use the same coordinate vectors over the full
  bound.
\end{theorem}

\begin{theorem}[The sphere presentation transports its bounded quotient products]
  \label{thm:computation-presentation-comparison-mul}
  \lean{KIP126.Classical.Adams.LinE2Presentation.comparison_mul,KIP126.Challenge2.ComputationBindings.presentation}
  \notready
  \uses{def:computation-fixed-bindings}
  For all $s,t,s',t'\in\mathbb N$ with $t+t'\le261$, every
  $x\in A(s,t)$, $y\in A(s',t')$ and
  $z\in A(s+s',t+t')$, if $|x|\,|y|=|z|$ in the same fixed quotient
  algebra, then
  \[
    P_{s+s',t+t'}(z)
      =P.\mathrm{product}_{s,t,s',t'}(P_{s,t}x,P_{s',t'}y).
  \]
  The sum-degree bound supplies the two input-degree bounds as well.
  This condition compares the quotient multiplication with the specified
  bilinear operation. Its agreement with the actual sphere-layer product
  is separately the condition of
  Theorem~\ref{thm:computation-sphereMultiplicative}.
\end{theorem}

\begin{theorem}[The bounded presentation product agrees with the prescribed sphere-layer product]
  \label{thm:computation-sphereMultiplicative}
  \lean{KIP126.Challenge2.SphereMultiplicativeInterface,KIP126.Challenge2.ComputationResults.sphereMultiplicative}
  \lean{KIP126.Classical.Adams.Sphere.Multiplication.unitSecondCycle,KIP126.Classical.Adams.Sphere.Multiplication.firstProduct}

    |z|=\mu_1(|a|,|a'|),\qquad
    [z]=P.\mathrm{product}_{s,t,s',t'}([a],[a']).
  \]
  Here $\mu_1$ is the already prescribed actual first-layer product
  formed from the fixed sphere tower pairing, tensor input and $H$-ring
  multiplication. The existential cycle is required to have exactly this
  underlying first-page value. This is the single bounded unit-and-product
  comparison condition; its mathematical certification remains required.
\end{theorem}

\begin{theorem}[Every selected product respects the fixed Milnor-coordinate sphere product]
  \label{thm:computation-route-products}
  \lean{KIP126.Computation.Route.CertifiedRealization.products,KIP126.Challenge2.ComputationResults.route,KIP126.Computation.Route.ProductCorrect}
  \notready
  \uses{def:computation-fixed-bindings}
  For every product record $p\in\mathrm{Raw.products}$, every
  $x\in A(p.s,p.t)$ and every $y\in A(p.sp,p.tp)$,
  \[
  \begin{aligned}
    &R.\mathrm{sphere}_{p.s+p.sp,p.t+p.tp}(\mathrm{mulAt}(x,y))\\
    &\qquad=\mu_{\mathrm{Mil}}
       (R.\mathrm{sphere}_{p.s,p.t}(x),R.\mathrm{sphere}_{p.sp,p.tp}(y)).
  \end{aligned}
  \]
  The left product is multiplication of homogeneous classes in the
  fixed polynomial-relation quotient. The right product is precisely
  \texttt{Sphere.Internal.product} for the fixed $H$ and Milnor
  cooperations $M$, constructed on the actual sphere $E_2$ through its
  fixed Milnor-cohomology comparison. This condition uses that operation;
  it does not replace it by $P.\mathrm{product}$.
\end{theorem}
