  $C_3$ is equivalent to the absence of a nonzero $d_6(W)=T$.

  $r=12$ and $D_{12}$ holds.  This conditional deduction neither proves

  branch under $C_3$; conversely the specified nonzero $d_{12}$ is an incoming
  hit.  All differentials used here are those of the fixed SSData object.
\end{proof}

\begin{theorem}[$h_6^2$ is a permanent cycle]
  \label{thm:h6-square-permanent}
  \lean{KIP126.Challenge.Final.H6SquarePermanent.h6_sq_permanent}
  \uses{def:h6-standard-internal-target}
  \notready
  The standard class $h_6^2\in E_2^{2,128}(S^0)$ is nonzero permanent
  in the classical mod--$2$ Adams spectral sequence of the sphere.
  Precisely, there exists a common $Z_\infty$ representative whose
  $E_2$ class is the standard $h_6^2$ and whose $E_\infty$ class is nonzero.
\end{theorem}

\begin{proof}
  \uses{prop:h6-internal-standard-comparison,prop:fixed-sphere-adams-applicability,prop:fixed-sphere-adams-separated,prop:possible-h-6-sq,prop:near126-c3-not-c5}
  Identify the standard square with the computational square using
  Proposition~\ref{prop:h6-internal-standard-comparison}. Apply the
  differential dichotomy and its characterization by $C_3,C_4,C_5$.
  If the nonzero
  $d_{12}$ alternative of
  Proposition~\ref{prop:possible-h-6-sq} held, then $C_3$, $C_4$, and $C_5$ would
  all hold.  Proposition~\ref{prop:near126-c3-not-c5} contradicts the first and
  third simultaneously.  Therefore the differential alternative is false,
  and the exclusive permanent-cycle alternative holds.
\end{proof}
